pythonimport re

def simple_sentiment_check(text):
    positive_words = ['good', 'great', 'happy', 'yes', 'agree', 'interested', 'excited', 'love', 'awesome', 'positive']
    negative_words = ['no', 'bad', 'expensive', 'doubt', 'not sure', 'frustrated', 'concerned', 'skeptical', 'negative', 'problem']
    
    text_lower = text.lower()
    pos_count = sum(1 for word in positive_words if re.search(r'\b' + word + r'\b', text_lower))
    neg_count = sum(1 for word in negative_words if re.search(r'\b' + word + r'\b', text_lower))
    
    if neg_count > pos_count:
        return 'negative', [word for word in negative_words if word in text_lower]
    elif pos_count > neg_count:
        return 'positive', [word for word in positive_words if word in text_lower]
    else:
        return 'neutral', []

# Test examples from sales context
examples = [
    "This seems too expensive for what we need.",  # Negative
    "I'm really excited about this solution!",     # Positive
    "Not sure if it's the right fit."              # Negative
]

results = {text: simple_sentiment_check(text) for text in examples}
print(results)

typescriptprivate performKeywordAnalysis(transcript: string): { type: 'positive' | 'neutral' | 'negative'; keywords: string[] } {
  const positiveKeywords = ['good', 'great', 'happy', 'yes', 'agree', 'interested', 'excited', 'love', 'awesome', 'positive'];
  const negativeKeywords = ['no', 'bad', 'expensive', 'doubt', 'not sure', 'frustrated', 'concerned', 'skeptical', 'negative', 'problem'];
  
  const textLower = transcript.toLowerCase();
  const foundPositive = positiveKeywords.filter(kw => textLower.includes(kw));
  const foundNegative = negativeKeywords.filter(kw => textLower.includes(kw));
  
  if (foundNegative.length > foundPositive.length) {
    return { type: 'negative', keywords: foundNegative };
  } else if (foundPositive.length > foundNegative.length) {
    return { type: 'positive', keywords: foundPositive };
  } else {
    return { type: 'neutral', keywords: [] };
  }
}
