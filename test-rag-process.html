// Test the RAG Process directly
async function testRAGProcess() {
    console.log('🧪 Starting RAG Process test...');
    
    // Import the processor
    const { automatedDocumentProcessor } = await import('./src/services/document/AutomatedDocumentProcessor.ts');
    
    // Test document content
    const testContent = `Never Split the Difference - Key Sales Techniques

TACTICAL EMPATHY
The foundation of all negotiation. Understanding the feelings and mindset of another in the moment and hearing what is behind those feelings so you increase your influence in all the moments that follow.

MIRRORING
Simply repeating the last three words (or the critical one to three words) of what someone has just said. This encourages them to keep talking and reveals more information.

LABELING
Spotting the other person's feelings and turning them into words. Use phrases like "It seems like..." or "It sounds like..." or "It looks like..."`;

    // Process the document
    const result = await automatedDocumentProcessor.processDocumentContent(
        testContent,
        'test-document.txt',
        {
            chunkSize: 512,
            chunkOverlap: 50,
            enableChromaDB: true,
            enableOllamaIntegration: true
        }
    );
    
    console.log('✅ RAG Process Result:', result);
    return result;
}

// Run the test
testRAGProcess().then(result => {
    console.log('🎉 Test completed successfully!', result);
}).catch(error => {
    console.error('❌ Test failed:', error);
});
